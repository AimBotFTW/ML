# PRACTICAL 08 - K-MEANS CLUSTERING
# Dataset: Mall_Customers.csv
# Method: Silhouette Method to select the number of clusters

import pandas as pd
import matplotlib.pyplot as plt

from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from sklearn.metrics import silhouette_score

# 1. Load the dataset
df = pd.read_csv("Mall_Customers.csv")

# 2. Select the two features used for clustering
X = df[["Annual Income (k$)", "Spending Score (1-100)"]]

# 3. Standardize the features
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 4. Find the optimal number of clusters using Silhouette Score
scores = []

for k in range(2, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    labels = model.fit_predict(X_scaled)
    scores.append(silhouette_score(X_scaled, labels))

# Select the K value with the highest Silhouette Score
k_values = list(range(2, 11))
optimal_k = k_values[scores.index(max(scores))]

# 5. Train the final K-Means model using the selected K
kmeans = KMeans(n_clusters=optimal_k, random_state=42, n_init=10)
clusters = kmeans.fit_predict(X_scaled)

# 6. Add cluster labels to the dataset
df["Cluster"] = clusters

print("Optimal number of clusters:", optimal_k)
print("Final Silhouette Score:", silhouette_score(X_scaled, clusters))
print("\nFirst five clustered records:")
print(df.head())
print("\nCluster sizes:")
print(df["Cluster"].value_counts().sort_index())

# 7. Visualize the final clusters
plt.figure(figsize=(8, 5))
plt.scatter(
    X_scaled[:, 0],
    X_scaled[:, 1],
    c=clusters,
    cmap="viridis",
    s=60
)
plt.xlabel("Annual Income (standardized)")
plt.ylabel("Spending Score (standardized)")
plt.title("K-Means Customer Clustering")
plt.show()
